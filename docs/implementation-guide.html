#!/usr/bin/env python3
"""
kubernetes/applications/sample-api/app.py
Lightweight zero-dependency HTTP microservice for testing the CI/CD + GitOps pipeline.
Exposes /healthz, /readyz, /metrics, and /api/v1/info on port 8080.
"""

import json
import os
import socket
from http.server import BaseHTTPRequestHandler, HTTPServer

APP_VERSION = os.environ.get("APP_VERSION", "1.0.0")
APP_ENV = os.environ.get("APP_ENV", "dev")


class ApiHandler(BaseHTTPRequestHandler):
    def _send_json(self, status_code: int, payload: dict) -> None:
        body = json.dumps(payload).encode("utf-8")
        self.send_response(status_code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self) -> None:
        if self.path in ("/healthz", "/readyz"):
            self._send_json(200, {"status": "ok", "version": APP_VERSION, "env": APP_ENV})
        elif self.path == "/metrics":
            metrics = (
                "# HELP api_up Whether the sample API service is healthy\n"
                "# TYPE api_up gauge\n"
                f'api_up{{env="{APP_ENV}",version="{APP_VERSION}"}} 1\n'
            ).encode("utf-8")
            self.send_response(200)
            self.send_header("Content-Type", "text/plain; version=0.0.4")
            self.send_header("Content-Length", str(len(metrics)))
            self.end_headers()
            self.wfile.write(metrics)
        else:
            self._send_json(
                200,
                {
                    "service": "api",
                    "version": APP_VERSION,
                    "environment": APP_ENV,
                    "hostname": socket.gethostname(),
                },
            )


if __name__ == "__main__":
    server = HTTPServer(("0.0.0.0", 8080), ApiHandler)
    print(f"Starting sample-api v{APP_VERSION} ({APP_ENV}) on :8080...")
    server.serve_forever()
#!/usr/bin/env python3
"""
kubernetes/applications/sample-api/test_app.py
Unit tests for app.py — verifies /healthz, /readyz, /metrics, and /api/v1/info.
"""

import json
import threading
import unittest
import urllib.request
from http.server import HTTPServer
import app as sample_app


class SampleApiTestCase(unittest.TestCase):
    @classmethod
    def setUpClass(cls) -> None:
        cls.server = HTTPServer(("127.0.0.1", 0), sample_app.ApiHandler)
        cls.port = cls.server.server_port
        cls.thread = threading.Thread(target=cls.server.serve_forever, daemon=True)
        cls.thread.start()

    @classmethod
    def tearDownClass(cls) -> None:
        cls.server.shutdown()
        cls.server.server_close()

    def _get(self, path: str):
        with urllib.request.urlopen(f"http://127.0.0.1:{self.port}{path}", timeout=3) as resp:
            return resp.status, resp.headers.get("Content-Type", ""), resp.read().decode("utf-8")

    def test_healthz_and_readyz(self) -> None:
        for endpoint in ("/healthz", "/readyz"):
            status, content_type, body = self._get(endpoint)
            self.assertEqual(status, 200)
            self.assertIn("application/json", content_type)
            self.assertEqual(json.loads(body)["status"], "ok")

    def test_prometheus_metrics(self) -> None:
        status, content_type, body = self._get("/metrics")
        self.assertEqual(status, 200)
        self.assertIn("api_up{", body)

    def test_service_info_payload(self) -> None:
        status, _, body = self._get("/api/v1/info")
        self.assertEqual(status, 200)
        self.assertEqual(json.loads(body)["service"], "api")


if __name__ == "__main__":
    unittest.main()

5. Git Push → Jenkins CI → Nexus → Argo CD GitOps Workflow Zero Cluster-Admin in Jenkins

Architectural Security Guardrail: Jenkins never holds Kubernetes cluster-admin credentials and never runs kubectl apply. Every deployment happens by Jenkins committing an immutable image tag to the GitLab devops/gitops repository, which Argo CD on Layer 2 observes and reconciles onto Layer 3.

Stage 1Git Push Webhook → Compute Immutable Tag 1.0.0-<sha>
Stage 2–3Install Deps & Run Unit Tests (JUnit XML)
Stage 4–5SonarQube Quality Gate + Trivy FS Scan
Stage 6–7Rootless Buildah Build + Trivy Image Scan
Stage 8Push Immutable Tags to Nexus :5000
Stage 9Commit newTag to devops/gitops → Argo CD Sync

Condition B: New Pod Crashes on Startup (CrashLoopBackOff / ImagePullBackOff) Bad env var, missing config, or startup exception; startupProbe / readinessProbe fails RollingUpdate uses maxUnavailable: 0: old healthy pods continue serving 100% of traffic while the new replica is stuck Revert the last GitOps overlay commit and re-sync:
./scripts/gitops-rollback.sh --app api --env production --revert-last		 < 30 s (Zero downtime)
Condition C: Post-Deploy 5xx Spike / Latency Regression Pods pass /readyz, but Prometheus alerts on HTTP 5xx > 1% or p95 latency regression Prometheus fires alert; Grafana Loki shows application stack traces Pin kustomization.yaml back to the last known-good immutable tag:
./scripts/gitops-rollback.sh --app api --env production --to-tag 1.0.0-a1b2c3d4		 < 60 s
Condition D: Bad ConfigMap, Secret, or Resource Limit Change Application image is fine, but a bad Kustomize patch or ConfigMap broke routing or caused OOMKilled Argo CD tracks full Git commit history across the entire overlay directory Revert the offending Git commit in devops/gitops (git revert <commit-sha> && git push), then argocd app sync api-production. < 60 s
Condition E: Bad Database Schema Migration + App Release New release ran a faulty SQL migration or caused lock contention on PostgreSQL HA PostgreSQL HA runs continuous WAL archiving + pgBackRest full/diff backups on Garage S3 1) Roll back app to previous tag via gitops-rollback.sh.
2) Run backward-compatible SQL down-migration (or pgbackrest --type=time restore for catastrophic data corruption).		 1–15 min
Condition F: Severe P1 Outage While GitLab is Unreachable Immediate production outage AND GitLab/CI plane is undergoing maintenance Kubernetes Deployment keeps revisionHistoryLimit: 10 ReplicaSets on Layer 3 1) Disable Argo CD auto-sync temporarily.
2) Run argocd app rollback api-production <id> or kubectl -n production rollout undo deploy/prod-api.
3) Backport the revert to devops/gitops as soon as GitLab is back.		 < 15 s
